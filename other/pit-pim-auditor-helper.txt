count(message) as count1, 
cast(count(message) as float)/cast((select count(1) from tmpcsvresult) as float) as ratio1
from tmpcsvresult t 
group by status, message 
;

-- miravia product publishing stats -- OK for info and miravia publish
select status, message, msg_miravia, 
count(msg_miravia) as count2, 
cast(count(msg_miravia) as float)/cast((select count(1) from tmpcsvresult) as float) as ratio2

where true 
and msg_miravia not like '%THD_IC_ERR_F_DOMAIN_IMAGE_00_01_003%' -- '%duplicate image%'
