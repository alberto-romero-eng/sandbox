### sql, create / truncate table

use pim;

CREATE TABLE tmpresult (
parent_sku text,
sku text,
gtin varchar(64),
itemId text,
status text,
message text,
PRIMARY KEY (gtin)
)
DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
;

truncate table tmpresult;


###
### sql, frequent queries using table tmpresult
###

use pim;

select * from tmpresult;

select count(1) from tmpresult;

select cast((select count(1) from tmpresult) as float);

select status, message, count(message), cast(count(message) as float)/cast((select count(1) from tmpresult) as float) 
from tmpresult t 
group by status, message 
;

select count(1) from
(
select tr.gtin, tr.status, tr.message, pe.gtin as pegtin, pe.data_source_id, pe.created_at 
from tmpresult tr 
left join product_error pe on tr.gtin = pe.gtin -- product_error may contain several times a gtin, thus left join (inner join would not make much sense)
where true 
and pe.gtin is null -- this filter always enabled, see comment regarding left join
and tr.status <> 'INFO_RECOVERED_CORRECTLY' -- check with / without this filter
-- and tr.message not like '%EAN%' -- check with / without this filter
) as t
;

select * from product_error ;
